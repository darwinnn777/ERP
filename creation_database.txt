-- 1. roles
CREATE TABLE roles (
  id INT NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
);

-- 2. users
CREATE TABLE users (
  id INT NOT NULL AUTO_INCREMENT,
  username VARCHAR(255) NOT NULL UNIQUE,
  full_name VARCHAR(255) DEFAULT NULL,
  password_hash VARCHAR(255) NOT NULL,
  role_id INT NOT NULL,
  PRIMARY KEY (id),
  CONSTRAINT users_role_fk FOREIGN KEY (role_id) REFERENCES roles(id)
);

-- 3. products
CREATE TABLE products (
  id INT NOT NULL AUTO_INCREMENT,
  sku VARCHAR(255) NOT NULL UNIQUE,
  name VARCHAR(255) NOT NULL,
  price_sell DECIMAL(10,2) NOT NULL,
  price_buy DECIMAL(10,2) NOT NULL,
  min_stock INT DEFAULT 10,
  product_type ENUM('Ingredient', 'Final Product', 'Both') DEFAULT 'Final Product',
  unit_of_measure VARCHAR(50) DEFAULT 'unit',
  image_url TEXT,
  is_discounted BOOLEAN DEFAULT FALSE,
  PRIMARY KEY (id)
);

-- 4. recipes
CREATE TABLE recipes (
  id INT NOT NULL AUTO_INCREMENT,
  final_product_id INT NOT NULL,
  ingredient_id INT NOT NULL,
  quantity_needed DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (id),
  CONSTRAINT fk_final_product FOREIGN KEY (final_product_id) REFERENCES products(id),
  CONSTRAINT fk_ingredient FOREIGN KEY (ingredient_id) REFERENCES products(id)
);

-- 5. purchase_orders
CREATE TABLE purchase_orders (
  id INT NOT NULL AUTO_INCREMENT,
  provider_name VARCHAR(255) NOT NULL,
  order_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  status ENUM('Pendiente', 'Recibido') DEFAULT 'Pendiente',
  total_amount DECIMAL(10,2) DEFAULT 0.00,
  PRIMARY KEY (id)
);

-- 6. purchase_order_items
CREATE TABLE purchase_order_items (
  id INT NOT NULL AUTO_INCREMENT,
  po_id INT,
  product_id INT,
  quantity DECIMAL(10,2) NOT NULL,
  price DECIMAL(10,2) NOT NULL DEFAULT 0.00,
  received BOOLEAN DEFAULT FALSE,
  PRIMARY KEY (id),
  CONSTRAINT purchase_order_items_po_id_fkey FOREIGN KEY (po_id) REFERENCES purchase_orders(id),
  CONSTRAINT purchase_order_items_product_id_fkey FOREIGN KEY (product_id) REFERENCES products(id)
);

-- 7. sales
CREATE TABLE sales (
  id INT NOT NULL AUTO_INCREMENT,
  total_amount DECIMAL(10,2) NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  payment_method ENUM('cash', 'card') NOT NULL,
  amount_paid DECIMAL(10,2) NOT NULL,
  change_amount DECIMAL(10,2) NOT NULL DEFAULT 0.00,
  PRIMARY KEY (id)
);

-- 8. sales_items
CREATE TABLE sales_items (
  id INT NOT NULL AUTO_INCREMENT,
  sale_id INT,
  product_id INT,
  quantity DECIMAL(10,2) NOT NULL,
  price DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (id),
  CONSTRAINT sales_items_sale_id_fkey FOREIGN KEY (sale_id) REFERENCES sales(id),
  CONSTRAINT sales_items_product_id_fkey FOREIGN KEY (product_id) REFERENCES products(id)
);

-- 9. warehouses
CREATE TABLE warehouses (
  id INT NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
);

-- 10. stock_lots
CREATE TABLE stock_lots (
  id INT NOT NULL AUTO_INCREMENT,
  product_id INT NOT NULL,
  warehouse_id INT NOT NULL,
  lot_number VARCHAR(255) NOT NULL,
  expiration_date DATE,
  quantity DECIMAL(10,2) NOT NULL DEFAULT 0.00,
  is_discounted BOOLEAN DEFAULT FALSE,
  PRIMARY KEY (id),
  CONSTRAINT stock_lots_product_fk FOREIGN KEY (product_id) REFERENCES products(id),
  CONSTRAINT stock_lots_warehouse_fk FOREIGN KEY (warehouse_id) REFERENCES warehouses(id)
);

-- 11. stock_movements
CREATE TABLE stock_movements (
  id INT NOT NULL AUTO_INCREMENT,
  product_id INT,
  quantity DECIMAL(10,2) NOT NULL,
  movement_type ENUM('IN', 'OUT'),
  reference_id INT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  CONSTRAINT stock_movements_product_id_fkey FOREIGN KEY (product_id) REFERENCES products(id)
);
